JAP_collaboration_on_CHI = {
	allowed = {
		tag = JAP
	}
	
	enable = {
		tag = JAP
		NOR = {
			exists = yes
			has_capitulated = no
			NOT = { is_ally_with = JAP }
		}
		# only try before end of 1941 (and continue if already doing)
		OR = {
			date < 1936.1.1
			is_preparing_operation = {
				operation = operation_collaboration_government
				target = CHI
			}
			is_running_operation = {
				operation = operation_collaboration_government
				target = CHI
			}
		}
		# do at most once or twice depending on date
		OR = {
			num_finished_operations = {
				operation = operation_collaboration_government
				target = CHI
				value < 1
			}
			AND = {
				date > 1941.1.1
				num_finished_operations = {
					operation = operation_collaboration_government
					target = CHI
					value < 2
				}
			}
		}
	}
	
	abort_when_not_enabled = yes
	
	ai_strategy = {
		type = operative_operation
		operation = operation_collaboration_government
		value = 600
		operation_target = CHI
	}
	
	ai_strategy = {
		type = operative_mission
		mission = build_intel_network
		value = 600
		mission_target = CHI
	}
}
