SOV_build_intel_network_in_MEX_strat = {
	allowed = {
		tag = SOV
	}
	
	enable = {
		tag = SOV
		OR = {
			AND = {
				date > 1940.1.1
				has_completed_focus = SOV_the_military_conspiracy #Parent focus of Behead the Snake
			}
			has_completed_focus = SOV_behead_the_snake
		}
	}
	
	abort = {
		has_idea = trotskyite_plot_purged
	}
		
	ai_strategy = {
		type = operative_mission
		mission = build_intel_network
		value = 1000
		mission_target = MEX
		num_operatives = 1
	}
}

		tag = SOV
	}
	
	enable = {
		has_global_flag = VUP_Hist_version
		tag = SOV
		date > 1937.6.1
		#NOT = { has_completed_focus = GER_the_four_year_plan }
		EST = {
			exists = yes
			has_capitulated = no
			NOT = { is_ally_with = SOV }
		}
		num_finished_operations = {
			operation = operation_collaboration_government
			target = EST
			value < 2
		}
	}

	abort = {

			target = EST
			value > 1

			target = EST
		}
	}

		type = operative_mission

		mission_target = EST
	}

		value = 1000
		operation_target = EST
	}

	ai_strategy = {
		type = intelligence_agency_usable_factories
		value = 50
	}
}

	}
}
