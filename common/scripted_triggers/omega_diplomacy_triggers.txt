# Запрещает запрос войск у стран, которыми управляет игрок
DIPLOMACY_REQUEST_EXP_FORCE_ENABLE_TRIGGER = {
    FROM = {
        is_ai = yes
    }
}

DIPLOMACY_GENERATE_WARGOAL_ENABLE_TRIGGER = {
    if = {
        limit = {
            has_country_flag = VUP_Hist_version
        }
        NOT = {has_country_flag = VUP_Hist_version}
    }
    
    if = {
        limit = {
            is_ai = no
        }
        OR = {
        # Оправдание на ИИ с 1937
        AND = {
            FROM = {
                is_ai = yes
            }
            date > 1938.1.1
        }
        # Оправдание на игрока с 1939
        AND = {
            FROM = { is_ai = no }
            date > 1939.1.1
        }
        hidden_trigger = {
        AND = {
            # Цель находится в Азии, Южной Америке или Австралии
            FROM = {
                OR = {
                    capital_scope = { is_on_continent = asia }
                    capital_scope = { is_on_continent = australia }
                    capital_scope = { is_on_continent = south_america }
                }
            }
            # Нападающий с одного из этих же континентов
            ROOT = {
                OR = {
                    capital_scope = { is_on_continent = asia }
                    capital_scope = { is_on_continent = australia }
                    capital_scope = { is_on_continent = south_america }
                }
            }
        }
        }
    }
    }
    
    custom_trigger_tooltip = {
		tooltip = vuina_diploma
	}

}

	}
}
